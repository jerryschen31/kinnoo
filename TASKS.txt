# The current task types are:

# Template (YAML):
# tasks:
#   - id: taskX
#     title: Short title
#     type: Task type (defaults to "task" if omitted)

#       - Step 1 description

#     dependencies: [taskZ]
#     tests: [test0, test1]
#     status: not-started / in-progress / etc

# A decision task has different fields:
#
# Template for decision tasks (YAML):
# tasks:
#   - id: taskX
#     title: Short title
#     type: decision
#     description: Description of the decision to be made. Can include some initial choices, if available.
#     options: Potential choices for the decision (e.g., choice of model), as a YAML style list. Populated once these are available.
#           - option1: overview/pros/cons/artifacts
#           - option2: overview/pros/cons/artifacts
#     criteria: list of measurable evaluation criteria (latency, accuracy, size, license)
#     selected_option: option id (set when status: approved)
#     rationale: short reasoning for selection.
#     approvals: list of approvers: human name / agent ID (optional)
#     status: not-started / in-progress / etc
